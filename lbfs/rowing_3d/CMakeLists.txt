cmake_minimum_required(VERSION 3.20)
project(policy7400_surface_loads LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
option(LU_PRECISE_DIVISION "Use precise CUDA float division instead of the validated performance option" OFF)
set(LU_DEFINITIONS LU_GPU_BLOCK_SIZE=128 LU_UPWIND_WENO_ONLY=1
  LU_SPECIALIZED_METRIC_FACES=1 LU_CACHED_STRESS_SOURCE=1 LU_PACKED_METRIC_GRADIENTS=1
  LU_INTERIOR_PRIMITIVE_PACK=1 LU_FUSE_SOURCE_DIVERGENCE=1 LU_CACHED_PRIMITIVES=1
  LU_SHARED_CAPILLARY_FACES=1)
include(CheckLanguage)
check_language(CUDA)
if(CMAKE_CUDA_COMPILER)
  enable_language(CUDA)
  add_executable(lu2025 src/main.cu)
  set_target_properties(lu2025 PROPERTIES CUDA_ARCHITECTURES 90 CUDA_STANDARD 17)
  target_compile_options(lu2025 PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-O3;--extended-lambda;--expt-relaxed-constexpr;-maxrregcount=128;--prec-sqrt=true;--ftz=false>)
  if(LU_PRECISE_DIVISION)
    target_compile_options(lu2025 PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--prec-div=true>)
  else()
    target_compile_options(lu2025 PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--prec-div=false>)
  endif()
  add_executable(buffer_lifetime tests/buffer_lifetime.cu)
  set_target_properties(buffer_lifetime PROPERTIES CUDA_ARCHITECTURES 90 CUDA_STANDARD 17)
  target_include_directories(buffer_lifetime PRIVATE src)
  target_compile_options(buffer_lifetime PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-O3;--extended-lambda;--expt-relaxed-constexpr>)
else()
  add_executable(lu2025 src/main.cpp)
  if(NOT MSVC)
    target_compile_options(lu2025 PRIVATE -O3)
  endif()
endif()
target_compile_definitions(lu2025 PRIVATE ${LU_DEFINITIONS})
add_executable(fluid_map_test tests/fluid_map.cpp)
target_compile_definitions(fluid_map_test PRIVATE ${LU_DEFINITIONS})
target_include_directories(fluid_map_test PRIVATE src)
add_executable(rk_scheme_test tests/rk_scheme_test.cpp)
target_include_directories(rk_scheme_test PRIVATE src)
add_executable(restart_clock_test tests/restart_clock.cpp)
target_include_directories(restart_clock_test PRIVATE src)
enable_testing()
add_executable(surface_force_test tests/surface_force_test.cpp)
target_include_directories(surface_force_test PRIVATE src)
add_test(NAME surface_force COMMAND surface_force_test)
add_executable(policy_interpolation_test tests/policy_interpolation_test.cpp)
target_include_directories(policy_interpolation_test PRIVATE src)
add_test(NAME rk_scheme COMMAND rk_scheme_test)
add_test(NAME restart_clock COMMAND restart_clock_test)
add_test(NAME fluid_map COMMAND fluid_map_test "${CMAKE_CURRENT_BINARY_DIR}/fluid_map.bin")
if(CMAKE_CUDA_COMPILER)
  add_test(NAME buffer_lifetime COMMAND buffer_lifetime)
  add_executable(surface_force_cuda tests/surface_force_test.cu)
  target_include_directories(surface_force_cuda PRIVATE src)
  set_target_properties(surface_force_cuda PROPERTIES CUDA_ARCHITECTURES 90 CUDA_STANDARD 17)
  target_compile_options(surface_force_cuda PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-O3;--extended-lambda;--expt-relaxed-constexpr>)
  add_test(NAME surface_force_gpu COMMAND surface_force_cuda)
endif()

target_include_directories(probe_routing_test PRIVATE src)
add_test(NAME probe_routing COMMAND probe_routing_test)
add_executable(checkpoint_store_test tests/checkpoint_store_test.cpp)
target_include_directories(checkpoint_store_test PRIVATE src)
add_test(NAME checkpoint_latest COMMAND checkpoint_store_test "${CMAKE_CURRENT_BINARY_DIR}/checkpoint_transaction_test")

  add_executable(probe_routing_cuda tests/probe_routing.cu)
