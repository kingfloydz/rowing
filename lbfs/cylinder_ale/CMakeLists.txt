cmake_minimum_required(VERSION 3.24)
project(lu2025_ale_single_gpu LANGUAGES CXX)
option(LU_ENABLE_CUDA "Build CUDA solver" OFF)
option(LU_DOUBLE "Use FP64" OFF)
set(LU_GPU_BLOCK_SIZE 64 CACHE STRING "CUDA threads per block")
set(LU_CUDA_MAXRREGCOUNT "148" CACHE STRING "H200 tuned register cap")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(LU_ENABLE_CUDA)
  enable_language(CUDA)
  add_executable(lu2025 src/main.cu)
  target_compile_options(lu2025 PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-O3;--extended-lambda;--expt-relaxed-constexpr;-lineinfo;-maxrregcount=${LU_CUDA_MAXRREGCOUNT}>)
  target_compile_definitions(lu2025 PRIVATE LU_GPU_BLOCK_SIZE=${LU_GPU_BLOCK_SIZE})
else()
  add_executable(lu2025 src/main.cpp)
  find_package(OpenMP)
  if(OpenMP_CXX_FOUND)
    target_link_libraries(lu2025 PRIVATE OpenMP::OpenMP_CXX)
  endif()
endif()
if(LU_DOUBLE)
  target_compile_definitions(lu2025 PRIVATE LU_DOUBLE=1)
endif()
if(NOT MSVC)
  target_compile_options(lu2025 PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-O3;-Wall;-Wextra;-Wpedantic>)
endif()
